Il vetro

Trasparenza = fill-opacity. Satinato = quanto sfochi ciò che sta dietro. 

<!-- vetro tinto: l'opacità del riempimento È la trasparenza scelta -->
<polygon
  :fill="options.glass.color.code"
  :fill-opacity="(options.glass.color.transparency ?? 0) / 100" />

<!-- satinato: sfoca ciò che si vede DIETRO il vetro -->
<filter id="satinationBlur">
  <feGaussianBlur :stdDeviation="satination * 5" />
</filter>
