Come si congela una doccia

Vitest + happy-dom. Che però non ha layout: tutto quello che dipende dal browser va fissato.

const renderSvg = async (inputs: Inputs) => {
  const wrapper = mount(ShowerPreviewerComponent, { props: { inputs } })
  await nextTick()

  // niente layout → fisso io la larghezza e sveglio il ResizeObserver (stub)
  Object.defineProperty(container, "clientWidth", { value: 800 })
  observerCallbacks.forEach((cb) => cb())
  await nextTick()

  return normalizeSvg(wrapper.find("#shower-preview-svg").element.outerHTML)
}

for (const { snapshot, inputs } of cases) {
  it(snapshot, async () => {
    await expect(await renderSvg(inputs)).toMatchFileSnapshot(`./__snapshots__/${snapshot}`)
  })
}
